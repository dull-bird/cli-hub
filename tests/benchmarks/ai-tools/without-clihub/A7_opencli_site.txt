我来帮你创建一个使用OpenCLI抓取B站热门视频的工具。我将使用Python和requests库来实现。

```python
#!/usr/bin/env python3
"""
Bilibili热门视频抓取工具
使用OpenCLI风格，支持多种热门分类抓取
"""

import requests
import json
import sys
import time
from typing import Optional, Dict, List, Any

class BilibiliHot:
    """Bilibili热门视频抓取器"""
    
    BASE_URL = "https://api.bilibili.com"
    
    def __init__(self, timeout: int = 10, delay: float = 0.5):
        self.session = requests.Session()
        self.timeout = timeout
        self.delay = delay
        self._setup_headers()
    
    def _setup_headers(self):
        """设置请求头"""
        self.session.headers.update({
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
                          "(KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
            "Referer": "https://www.bilibili.com/",
            "Accept": "application/json, text/plain, */*",
            "Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
        })
    
    def _make_request(self, url: str, params: Optional[Dict] = None) -> Dict:
        """发送请求并返回JSON数据"""
        try:
            resp = self.session.get(url, params=params, timeout=self.timeout)
            resp.raise_for_status()
            data = resp.json()
            
            if data.get("code") != 0:
                print(f"[警告] API返回错误: {data.get('message', '未知错误')}", file=sys.stderr)
                return {}
            
            return data.get("data", {})
        except requests.exceptions.Timeout:
            print(f"[错误] 请求超时: {url}", file=sys.stderr)
        except requests.exceptions.RequestException as e:
            print(f"[错误] 请求失败: {e}", file=sys.stderr)
        except json.JSONDecodeError:
            print(f"[错误] JSON解析失败", file=sys.stderr)
        
        return {}
    
    def get_popular(self, pn: int = 1, ps: int = 50) -> List[Dict[str, Any]]:
        """
        获取热门视频列表
        
        Args:
            pn: 页码 (从1开始)
            ps: 每页数量 (最大50)
        
        Returns:
            视频列表
        """
        url = f"{self.BASE_URL}/x/web-interface/popular"
        params = {"pn": pn, "ps": min(ps, 50)}
        
        data = self._make_request(url, params)
        videos = data.get("list", [])
        
        # 添加延迟，避免请求过快
        time.sleep(self.delay)
        
        return videos
    
    def get_hot_words(self) -> List[str]:
        """获取热门搜索词"""
        url = f"{self.BASE_URL}/x/web-interface/hot"
        data = self._make_request(url)
        return [item["keyword"] for item in data.get("list", [])]
    
    def get_popular_series(self) -> List[Dict[str, Any]]:
        """获取热门系列列表"""
        url = f"{self.BASE_URL}/x/web-interface/popular/series/list"
        data = self._make_request(url)
        return data.get("list", [])
    
    def format_video_info(self, video: Dict[str, Any], index: int) -> str:
        """格式化视频信息"""
        # 基本信息
        title = video.get("title", "未知标题")
        author = video.get("owner", {}).get("name", "未知UP主")
        bvid = video.get("bvid", "未知ID")
        
        # 统计信息
        stat = video.get("stat", {})
        view_count = self._format_number(stat.get("view", 0))
        like_count = self._format_number(stat.get("like", 0))
        danmaku_count = self._format_number(stat.get("danmaku", 0))
        
        # 时长
        duration = video.get("duration", 0)
        duration_str = f"{duration // 60}:{duration % 60:02d}"
        
        # 标签
        tname = video.get("tname", "未知分类")
        
        # 发布时间
        pubdate = video.get("pubdate", 0)
        pubdate_str = time.strftime("%Y-%m-%d %H:%M", time.localtime(pubdate))
        
        return (
            f"{'='*60}\n"
            f"  #{index} | {tname} | ⏱ {duration_str}\n"
            f"  标题: {title}\n"
            f"  UP主: {author} | BV号: {bvid}\n"
            f"  👁 {view_count} | 👍 {like_count} | 💬 {danmaku_count}\n"
            f"  发布时间: {pubdate_str}\n"
            f"  https://www.bilibili.com/video/{bvid}\n"
        )
    
    @staticmethod
    def _format_number(num: int) -> str:
        """格式化数字显示"""
        if num >= 100000000:
            return f"{num / 100000000:.1f}亿"
        elif num >= 10000:
            return f"{num / 10000:.1f}万"
        else:
            return str(num)
    
    def print_popular(self, page: int = 1, count: int = 20):
        """打印热门视频列表"""
        print(f"\n📺 B站热门视频 (第{page}页)\n")
        
        videos = self.get_popular(pn=page, ps=count)
        
        if not videos:
            print("暂无数据，请稍后重试")
            return
        
        for i, video in enumerate(videos, 1):
            print(self.format_video_info(video, (page - 1) * count + i))
        
        print(f"\n共 {len(videos)} 条数据\n")
    
    def print_hot_words(self):
        """打印热门搜索词"""
        print("\n🔥 B站热门搜索词\n")
        words = self.get_hot_words()
        
        if not words:
            print("暂无数据")
            return
        
        for i, word in enumerate(words[:20], 1):
            print(f"  {i:2d}. {word}")
        print()

    def print_series(self):
        """打印热门系列"""
        print("\n📋 B站热门系列\n")
        series_list = self.get_popular_series()
        
        if not series_list:
            print("暂无数据")
            return
        
        for i, series in enumerate(series_list[:10], 1):
            name = series.get("name", "未知系列")
            status = "进行中" if series.get("status") == 2 else "已结束"
            print(f"  {i:2d}. {name} [{status}]")
        print()


def main():
    """主函数 - CLI入口"""
    import argparse
    
    parser = argparse.ArgumentParser(
        description="Bilibili热门视频抓取工具",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
使���示例:
  %(prog)s                    # 获取热门视频 (默认20条)
  %(prog)s -p 2 -c 30        # 获取第2页，30条
  %(prog)s --hot-words       # 查看热门搜索词
  %(prog)s --series          # 查看热门系列
  %(prog)s --json -c 5       # 输出JSON格式
        """
    )
    
    parser.add_argument("-p", "--page", type=int, default=1, help="页码 (默认: 1)")
    parser.add_argument("-c", "--count", type=int, default=20, help="每页数量 (默认: 20, 最大: 50)")
    parser.add_argument("--hot-words", action="store_true", help="显示热门搜索词")
    parser.add_argument("--series", action="store_true", help="显示热门系列")
    parser.add_argument("--json", action="store_true", help="输出JSON格式")
    parser.add_argument("--delay", type=float, default=0.5, help="请求间隔秒数 (默认: 0.5)")
    
    args = parser.parse_args()
    
    # 创建抓取器
    fetcher = BilibiliHot(delay=args.delay)
    
    try:
        if args.hot_words:
            if args.json:
                words = fetcher.get_hot_words()
                print(json.dumps(words, ensure_ascii=False, indent=2))
            else:
                fetcher.print_hot_words()
        elif args.series:
            if args.json:
                series = fetcher.get_popular_series()
                print(json.dumps(series, ensure_ascii=False, indent=2))
            else:
                fetcher.print_series()
        else:
            if args.json:
                videos = fetcher.get_popular(pn=args.page, ps=args.count)
                print(json.dumps(videos, ensure_ascii=False, indent=2))
            else:
                fetcher.print_popular(page=args.page, count=args.count)
    
    except KeyboardInterrupt:
        print("\n\n已取消")
        sys.exit(0)


if __name__ == "__main__":
    main()
```

## 使用说明

### 安装依赖
```bash
pip install requests
```

### 基础用法

1. **获取热门视频（默认20条）**
```bash
python bilibili_hot.py
```

2. **获取指定页数和数量**
```bash
python bilibili_hot.py -p 2 -c 30
```

3. **查看热门搜索词**
```bash
python bilibili_hot.py --hot-words
```

4. **查看热门系列**
```bash
python bilibili_hot.py --series
```

5. **输出JSON格式**
```bash
python bilibili_hot.py --json -c 5
```

### CLI参数说明

| 参数 | 说明 |
|------|------|
| `-p, --page` | 页码，默认1 |
| `-c, --count` | 每页数量，默认20，最大50 |
| `--hot-words` | 显示热门搜索词 |
| `--series` | 显示热门系列 |
| `--json` | 输出JSON格式 |
| `--delay` | 请求间隔秒数，默认0.5 |

这个工具使用B站公开API，不需要登录即可使用。注意不要过于频繁请求，默认已经设置了0.5秒的延迟。
